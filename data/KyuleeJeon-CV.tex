\section{Summary}
Ph.D. candidate who makes \textbf{multimodal health data interoperable} so that clinical AI can be validated and adopted in routine care.
My research (i) \textbf{links medical images, waveforms, and electronic health records} in one standardized data model, as a co-author of the Medical Imaging CDM (the imaging extension for OMOP CDM);
(ii) measures the \textbf{real-world data heterogeneity} that limits data reuse and AI deployment; and
(iii) \textbf{generates evidence for clinical AI} through multi-institutional, distributed validation.
Seven peer-reviewed journal articles, five as first or co-first author. Lead organizer of the 2026 OHDSI APAC Datathon, with active collaborations with many institutions in Korea and abroad.

\section{Selected Research}
\textbf{Multimodal Data Interoperability: Imaging Extension for OMOP CDM (MI-CDM)}\rt{2022--present}\\
\textit{with Johns Hopkins University and the OHDSI Medical Imaging Working Group}
\begin{itemize}
\item \textbf{Review article} (\textit{first author}, Invest Radiol 2025): reviews how standardized imaging data, linked with clinical data in a common data model, supports rapid AI development, rigorous validation, and reproducibility.
\item \textbf{MI-CDM} (\textit{co-first author}, J Imaging Inform Med 2024): co-authored the imaging extension for OMOP CDM. Two extension tables link imaging events and image-derived features with clinical data.
\item \textbf{MIMIC-ECG to MI-CDM}: ETL of MIMIC-IV ECG data into the extension tables. \href{https://github.com/kyulee-jeon/MIMIC-ECG}{github.com/kyulee-jeon/MIMIC-ECG}
\item \textbf{MIMIC-CXR to MI-CDM}: ETL of MIMIC chest X-ray data into the extension tables. \href{https://github.com/kyulee-jeon/MIMIC-CXR}{github.com/kyulee-jeon/MIMIC-CXR}
\item \textbf{End-to-end AI validation pipeline} on this data model: case study with chest X-rays of tuberculosis patients (OHDSI Global Symposium 2026).
\item \textbf{Multi-institutional validation network} (\textit{project manager}; funded by a Seoul National University donation fund): built the same data structure with Seoul National University Hospital and SMG-SNU Boramae Medical Center, and tested the AI validation pipeline across the institutions.
\end{itemize}
\vspace{2pt}
\textbf{Real-World DICOM Metadata Heterogeneity}\rt{2024--present}
\begin{itemize}
\item \textbf{Nationwide network study} (\textit{first author}, IEEE J Biomed Health Inform, major revision): characterizes DICOM metadata heterogeneity in radiology across institutions.
\item \textbf{DICOM compliance assessment}: compliance with mandatory DICOM attributes \aw{KOSMI 2024 Excellence Prize}.
\end{itemize}
\vspace{2pt}
\textbf{Clinical AI Development and Terminology Standardization}\rt{2022--present}
\begin{itemize}
\item \textbf{ARKE} (\textit{co-first author}, J Biomed Inform 2026): ontology-driven LLM framework that maps local radiology procedure names to the LOINC/RSNA Radiology Playbook; validated at two tertiary hospitals.
\item \textbf{STEMI detection} (\textit{co-first author}, Ann Emerg Med 2024): deep ensemble ECG model with clinical validation \aw{KOSMI 2022 Excellence Prize}.
\end{itemize}
